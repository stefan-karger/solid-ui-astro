## Usage

```tsx
import { Label } from "~/components/ui/label"
```

```tsx
<Label for="email">Email address</Label>
```

## Label in Field

For form inputs, place the label directly above the related control.

```tsx showLineNumbers
<div class="grid gap-2">
  <Label for="email">Email address</Label>
  <Input id="email" type="email" />
</div>
```
