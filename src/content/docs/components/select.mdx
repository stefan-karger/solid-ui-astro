## Usage

```tsx showLineNumbers
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue
} from "~/components/ui/select"
```

```tsx showLineNumbers
type Fruit = {
  label: string
  value: string
}

const fruits: Fruit[] = [
  { label: "Apple", value: "apple" },
  { label: "Banana", value: "banana" },
  { label: "Blueberry", value: "blueberry" }
]

<Select<Fruit>
  options={fruits}
  optionValue="value"
  optionTextValue="label"
  placeholder="Select a fruit"
  itemComponent={(props) => (
    <SelectItem item={props.item}>{props.item.rawValue.label}</SelectItem>
  )}
>
  <SelectTrigger class="w-full max-w-48">
    <SelectValue<Fruit> />
  </SelectTrigger>
  <SelectContent />
</Select>
```

## Grouped Options

Use `optionGroupChildren` with `sectionComponent` to render grouped options with labels and separators.

```tsx showLineNumbers
<Select<Item, Group>
  options={groups}
  optionGroupChildren="options"
  optionValue="value"
  optionTextValue="label"
  sectionComponent={(props) => (
    <>
      <Show when={props.section.index !== 0}>
        <SelectSeparator />
      </Show>
      <SelectGroup>
        <SelectLabel>{props.section.rawValue.label}</SelectLabel>
      </SelectGroup>
    </>
  )}
  itemComponent={(props) => <SelectItem item={props.item}>{props.item.rawValue.label}</SelectItem>}
/>
```

## Alignment

Use `placement` on `Select` to control where the dropdown content appears relative to the trigger.

```tsx showLineNumbers
<Select placement="bottom-start">
  <SelectTrigger>
    <SelectValue placeholder="Select an option" />
  </SelectTrigger>
  <SelectContent />
</Select>
```

## Invalid State

Set `validationState="invalid"` on `Select` and `aria-invalid` on `SelectTrigger` to show error styling.

```tsx showLineNumbers /validationState="invalid"/ /aria-invalid/
<Select validationState="invalid">
  <SelectTrigger aria-invalid>
    <SelectValue placeholder="Select a fruit" />
  </SelectTrigger>
  <SelectContent />
</Select>
```

## API Reference

### Core Parts

| Part                 | Description                                                        |
| -------------------- | ------------------------------------------------------------------ |
| `Select`             | Root component that manages state, collection, and popup behavior. |
| `SelectTrigger`      | Button element that toggles and anchors the popup.                 |
| `SelectValue`        | Displays placeholder text or the selected option label.            |
| `SelectContent`      | Popup container rendered in a portal.                              |
| `SelectGroup`        | Section wrapper for grouped options.                               |
| `SelectLabel`        | Group label shown inside content.                                  |
| `SelectItem`         | Selectable option row.                                             |
| `SelectSeparator`    | Visual separator between groups.                                   |
| `SelectHiddenSelect` | Hidden native select for form integration.                         |

### Select

| Prop                  | Type                                  |
| --------------------- | ------------------------------------- |
| `options`             | `T[]`                                 |
| `optionValue`         | `keyof T \| ((option: T) => string)`  |
| `optionTextValue`     | `keyof T \| ((option: T) => string)`  |
| `optionDisabled`      | `keyof T \| ((option: T) => boolean)` |
| `optionGroupChildren` | `keyof G \| ((group: G) => T[])`      |
| `value`               | `T`                                   |
| `defaultValue`        | `T`                                   |
| `onChange`            | `(value: T) => void`                  |
| `placeholder`         | `string`                              |
| `placement`           | `"bottom" \| "bottom-start" \| ...`   |
| `disabled`            | `boolean`                             |
| `validationState`     | `"valid" \| "invalid"`                |

### SelectTrigger

| Prop    | Type              | Default     |
| ------- | ----------------- | ----------- |
| `size`  | `"sm"\|"default"` | `"default"` |
| `class` | `string`          | -           |

`SelectDescription` and `SelectErrorMessage` are also exported for accessible form hints and error messages.

See the [Kobalte Select](https://kobalte.dev/docs/core/components/select) documentation
for complete behavior and all supported props.
